BASIC USAGE
-----------
#include "cnn_denoise.h"

// 1. Create config with defaults
CNNConfig cfg = cnn_default_config(256, 256, 4);

// 2. Create network
CNNDenoiser* cnn = cnn_create(cfg);

// 3. Add layers
cnn_add_layer(cnn, (LayerConfig){4, 16, 1, "enc"});
cnn_add_layer(cnn, (LayerConfig){16, 4, 1, "dec"});
cnn_finalize(cnn);

// 4. Train
float loss = cnn_train_step(cnn, noisy, target, 1);

// 5. Cleanup
cnn_destroy(cnn);


ENABLE RESIDUAL MODE (recommended for denoising)
-------------------------------------------------
cfg.residual_mode = 1;

Note: Target should be NOISE, not clean image
  noise_target[i] = noisy[i] - clean[i];


USE ADAM OPTIMIZER (faster convergence)
---------------------------------------
cfg.optimizer = OPTIMIZER_ADAM;
cfg.learning_rate = 0.001f;


ADD EDGE PRESERVATION (better quality)
---------------------------------------
cfg.loss_config.num_losses = 2;
cfg.loss_config.types[0] = LOSS_MAE;
cfg.loss_config.weights[0] = 1.0f;
cfg.loss_config.types[1] = LOSS_LAPLACE;
cfg.loss_config.weights[1] = 0.1f;


LEARNING RATE DECAY
-------------------
// Exponential decay: lr * decay_rate^(step/decay_steps)
LearningRateDecay lr_decay;
learning_rate_decay_init(&lr_decay, 0.001f, 0.95f, 100);

for (int i = 0; i < 1000; i++) {
    float loss = cnn_train_step(cnn, noisy, target, 1);
    float new_lr = learning_rate_decay_get(&lr_decay, i);
    cnn_set_learning_rate(cnn, new_lr);
}


OPTIMIZER COMPARISON
--------------------
Feature              | SGD            | Adam
---------------------|----------------|----------------
Speed per iteration  | 7.3ms (fast)   | 11.3ms (slower)
Convergence speed    | Slow           | Fast
Memory usage         | Low            | High (4x params)
Recommended LR       | 0.0001         | 0.001
Best for             | Fine-tuning    | Initial training


LOSS FUNCTION GUIDE
-------------------
Loss Type       | Use Case              | Typical Weight
----------------|----------------------|----------------
LOSS_MAE        | General denoising     | 1.0
LOSS_MSE        | Smooth results        | 1.0
LOSS_LAPLACE    | Edge preservation     | 0.1

Recommended combinations:
- Sharp images: MAE (1.0) + Laplace (0.1)
- Smooth images: MSE (1.0)
- Balanced: MAE (0.7) + MSE (0.2) + Laplace (0.1)

COMPLETE EXAMPLE (all features)
--------------------------------
CNNConfig cfg = cnn_default_config(256, 256, 4);
cfg.residual_mode = 1;
cfg.optimizer = OPTIMIZER_ADAM;
cfg.learning_rate = 0.001f;
cfg.loss_config.num_losses = 2;
cfg.loss_config.types[0] = LOSS_MAE;
cfg.loss_config.weights[0] = 1.0f;
cfg.loss_config.types[1] = LOSS_LAPLACE;
cfg.loss_config.weights[1] = 0.1f;

CNNDenoiser* cnn = cnn_create(cfg);
cnn_add_layer(cnn, (LayerConfig){4, 16, 1, "enc1"});
cnn_add_layer(cnn, (LayerConfig){16, 32, 1, "enc2"});
cnn_add_layer(cnn, (LayerConfig){32, 16, 1, "dec1"});
cnn_add_layer(cnn, (LayerConfig){16, 4, 1, "dec2"});
cnn_finalize(cnn);

// Training loop
for (int epoch = 0; epoch < 100; epoch++) {
    float loss = cnn_train_step(cnn, noisy, noise_target, 1);
    printf("Epoch %d: Loss = %.6f\n", epoch, loss);
}

cnn_destroy(cnn);


TROUBLESHOOTING
---------------
Q: Training loss not decreasing?
A: Try Adam optimizer with higher LR (0.001)

Q: Results too smooth, losing detail?
A: Add Laplace loss with weight 0.1-0.3

Q: Training too slow?
A: Use SGD instead of Adam, or reduce image size

Q: Getting NaN losses?
A: Reduce learning rate by 10x
